```datagrok-exec
// Describe the molecule column
const molCol = t.columns.bySemType(DG.SEMTYPE.MOLECULE);

let descResult;
if (!molCol) {
  descResult = ui.divText('No molecule column found in the current table.');
} else {
  const s = molCol.stats;
  descResult = ui.tableFromMap({
    'Name': molCol.name,
    'Type': molCol.type,
    'Semantic Type': molCol.semType ?? '(none)',
    'Length': molCol.length,
    'Missing Values': s.missingValueCount,
    'Unique Values': s.uniqueCount,
    'Units': molCol.meta.units ?? '(none)',
    'Description': molCol.meta.description ?? '(none)',
  });
}

// Add a Ki column with semType Ki
const kiName = t.columns.getUnusedName('Ki');
const kiCol = t.columns.addNewFloat(kiName);
kiCol.semType = DG.SEMTYPE.Ki;
kiCol.meta.units = 'nM';

return ui.divV([
  ui.h2(molCol ? `Molecule column: "${molCol.name}"` : 'Molecule Column'),
  descResult,
  ui.divText(`✓ Added column "${kiName}" with semType Ki (units: nM)`),
]);
```
