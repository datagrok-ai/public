const smilesCols = grokky.findColumns(t, {semType: DG.SEMTYPE.MOLECULE, fuzzy: true});
const activityCols = grokky.findColumns(t, {name: 'activity', fuzzy: true});

if (!smilesCols.length) throw new Error('No SMILES/molecule column found.');
if (!activityCols.length) throw new Error('No activity column found.');

const smilesName = smilesCols[0].column.name;
const activityName = activityCols[0].column.name;

const copy = grokky.cloneDf(t, {cols: [smilesName, activityName], rows: 'filtered'});
copy.name = `${t.name}_filtered_copy`;

grok.shell.addTableView(copy);
