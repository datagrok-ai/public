```datagrok-exec
// Query ChEMBL compounds active against Thrombin (CHEMBL204 - Coagulation factor II)
// Using the organism query as the available entry point
const df = await grok.functions.call('Chembl:getChemblCompoundsByOrganism', {
  maxNumberOfMolecules: 1000,
  organism: 'Homo sapiens'
});

grok.shell.addTableView(df);

const rowCount = df.rowCount;
const colNames = df.columns.names();

// Try to filter for thrombin if a target column exists
let thrombinDf = null;
for (const colName of colNames) {
  const col = df.col(colName);
  if (col && col.type === DG.TYPE.STRING) {
    const sample = col.get(0);
    if (typeof sample === 'string' && (colName.toLowerCase().includes('target') || colName.toLowerCase().includes('pref'))) {
      // Filter for thrombin
      const mask = DG.BitSet.create(df.rowCount);
      for (let i = 0; i < df.rowCount; i++) {
        const val = col.get(i);
        if (val && val.toLowerCase().includes('thrombin')) mask.set(i, true);
      }
      if (mask.trueCount > 0) {
        thrombinDf = df.clone(mask);
        grok.shell.addTableView(thrombinDf);
        break;
      }
    }
  }
}

return ui.tableFromMap({
  'Total compounds loaded': rowCount,
  'Thrombin-specific hits': thrombinDf ? thrombinDf.rowCount : 'Target column not found for filtering',
  'Columns available': colNames.join(', ')
});
```
