```datagrok-exec
const df = await grok.functions.call('MolTrack:advancedSearch', {
  outputFields: [
    'compounds.canonical_smiles',
    'compounds.corporate_compound_id',
    'compounds.details.MW',
    'compounds.details.project'
  ],
  filter: {
    operator: 'AND',
    conditions: [
      {
        field: 'compounds.details.MW',
        operator: 'RANGE',
        value: [200, 500]
      },
      {
        field: 'compounds.details.project',
        operator: '=',
        value: 'Kinase'
      }
    ]
  }
});

if (!df || df.rowCount === 0)
  return ui.divText('No compounds found matching MW 200–500 and project "Kinase".');

grok.shell.addTableView(df);
return DG.Viewer.grid(df).root;
```

- **MW RANGE [200, 500]** — filters compounds with molecular weight between 200 and 500 Da
- **project = 'Kinase'** — restricts to the `Kinase` project
