const nameCol = DG.Column.fromStrings('names', ['ibuprofen']);
const namesDf = DG.DataFrame.fromColumns([nameCol]);

// namesToSmiles requires a dataframe + column
const smilesDf = await grok.functions.call('Chem:namesToSmiles', {
  data: namesDf,
  names: nameCol
});

// Extract the SMILES string
const smilesCol = smilesDf.columns.byIndex(0);
const smiles = smilesCol.get(0);

// Step 2: Compute MW and LogP
const molCol = DG.Column.fromStrings('molecule', [smiles]);
molCol.semType = DG.SEMTYPE.MOLECULE;
const molDf = DG.DataFrame.fromColumns([molCol]);

const propsDf = await grok.functions.call('Chem:getProperties', {
  molecules: molCol,
  selected: ['MW', 'LogP']
});

const mw = propsDf.col('MW') ? propsDf.col('MW').get(0) : propsDf.columns.byIndex(0).get(0);
const logp = propsDf.col('LogP') ? propsDf.col('LogP').get(0) : propsDf.columns.byIndex(1).get(0);

return ui.tableFromMap({

  'SMILES': smiles,

  'LogP': logp?.toFixed ? logp.toFixed(3) : logp,
});
