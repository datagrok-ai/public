```datagrok-exec
const result = await grok.functions.call('Reinvent4:reinvent', {
  ligand: 'CCN(CC)CC',
  optimize: 'BACE1'
});

// Show top 10 rows
const top10 = result.rows.match(Array.from({length: Math.min(10, result.rowCount)}, (_, i) => i));
const df10 = result.clone(top10);

return DG.Viewer.grid(df10).root;
```
