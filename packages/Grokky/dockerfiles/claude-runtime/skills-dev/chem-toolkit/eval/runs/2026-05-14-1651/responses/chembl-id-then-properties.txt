```datagrok-exec
// Step 1: Create a small DataFrame with the ChEMBL ID and look up its SMILES
const idDf = DG.DataFrame.fromColumns([
  DG.Column.fromStrings('chembl_id', ['CHEMBL1234'])
]);

// Map ChEMBL ID → SMILES
const smilesDF = await grok.functions.call('Chem:mapIdentifiersTransform', {
  table: idDf,
  molecules: idDf.col('chembl_id'),
  fromSource: 'chembl',
  toSource: 'smiles'
});

// Get the SMILES value
const smilesCol = idDf.columns.toList().find(c => c.name.toLowerCase().includes('smiles'));
const smiles = smilesCol ? smilesCol.get(0) : null;

if (!smiles) {
  return ui.divText('Could not resolve SMILES for CHEMBL1234. Please check the ChEMBL ID.');
}

// Step 2: Build a molecule DataFrame and compute descriptors
const molDf = DG.DataFrame.fromColumns([
  DG.Column.fromStrings('molecule', [smiles])
]);
molDf.columns.byIndex(0).semType = DG.SEMTYPE.MOLECULE;

const descDf = await grok.functions.call('Chem:getDescriptors', {
  molecules: molDf.col('molecule')
});

// Show a summary card first
const card = ui.tableFromMap({ 'ChEMBL ID': 'CHEMBL1234', 'SMILES': smiles });

return ui.divV([
  ui.h2('CHEMBL1234 — Molecular Descriptors'),
  card,
  ui.h3('RDKit Descriptors'),
  DG.Viewer.grid(descDf).root
]);
```
