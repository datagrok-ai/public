const grid = view.grid;

for (let i = 1; i < grid.columns.length; i++) {
  const gc = grid.columns.byIndex(i);
  if (!gc) continue;
  const col = t.col(gc.name);
  if (col && (col.semType === 'Molecule' || col.name.toLowerCase().includes('smiles') || col.name.toLowerCase().includes('mol'))) {
    gc.width = 300;
    found = true;
    break;
  }
}

if (!found) {
