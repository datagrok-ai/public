```datagrok-exec
const col = t.columns.bySemType(DG.SEMTYPE.MOLECULE);
if (!col) throw new Error('No molecule column found in the current table');

const query = 'CCO';
const molBlock = DG.chem.convert(query, DG.chem.Notation.Smiles, DG.chem.Notation.MolBlock);

view.getFiltersGroup({createDefaultFilters: false}).updateOrAdd({
  type: DG.FILTER_TYPE.SUBSTRUCTURE,
  column: col.name,
  columnName: col.name,
  molBlock,
  molBlockFailover: query,
});
```

I've applied a substructure filter for **CCO** (ethanol / hydroxylethyl fragment) on the molecule column. Only rows containing the CCO substructure are now visible. The filter widget also appears in the filter panel, where you can modify or remove it as needed.
