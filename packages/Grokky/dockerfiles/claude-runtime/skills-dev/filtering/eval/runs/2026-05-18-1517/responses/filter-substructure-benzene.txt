```datagrok-exec
// Find the molecule column
const molCol = t.columns.bySemType(DG.SEMTYPE.MOLECULE);
if (!molCol) throw new Error('No molecule column found in the current table');

const query = 'c1ccccc1'; // benzene SMILES
const molBlock = DG.chem.convert(query, DG.chem.Notation.Smiles, DG.chem.Notation.MolBlock);

view.getFiltersGroup({createDefaultFilters: false}).updateOrAdd({
  type: DG.FILTER_TYPE.SUBSTRUCTURE,
  column: molCol.name,
  columnName: molCol.name,
  molBlock,
  molBlockFailover: query,
});
```

I've applied a **benzene ring substructure filter** on the molecule column (`c1ccccc1`). The filter panel will show a substructure filter widget, and only rows containing a benzene ring will be visible. You can remove or adjust it from the filter panel at any time.
