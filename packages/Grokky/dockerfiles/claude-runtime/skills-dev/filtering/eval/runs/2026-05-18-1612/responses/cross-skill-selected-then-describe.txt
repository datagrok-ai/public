const rows = t.filter.trueCount;
const cols = t.columns.toList();

// Find molecule column (semType = Molecule, or common name patterns)

// Build a summary
const summary = {};
summary['Visible rows (filtered)'] = rows;
summary['Total columns'] = cols.length;

if (molCol) {
  summary['Molecule column'] = molCol.name;
}

// Summarize numeric columns
for (const col of cols) {
  if (col.type === DG.COLUMN_TYPE.FLOAT || col.type === DG.COLUMN_TYPE.INT) {
    const stats = col.stats;
    summary[`${col.name} (min)`] = stats.min?.toFixed(2) ?? 'N/A';
    summary[`${col.name} (max)`] = stats.max?.toFixed(2) ?? 'N/A';
    summary[`${col.name} (avg)`] = stats.avg?.toFixed(2) ?? 'N/A';
  }
}

// Summarize categorical columns
for (const col of cols) {
  if (col.type === DG.COLUMN_TYPE.STRING && col.categories.length <= 20) {
    summary[`${col.name} (unique values)`] = col.categories.length;
  }
}

return ui.tableFromMap(summary);
