const selMask = t.selection.clone();
t.filter.copyFrom(selMask);

const visCount = t.filter.trueCount;

// Step 2: describe the filtered rows
const stats = {};
stats['Filtered rows (selected)'] = visCount;

for (let i = 0; i < t.columns.length; i++) {
  const col = t.columns.byIndex(i);
  if (col.type === DG.COLUMN_TYPE.FLOAT || col.type === DG.COLUMN_TYPE.INT) {
    const vals = [];
    for (let r = 0; r < t.rowCount; r++) {
      if (t.filter.get(r)) vals.push(col.get(r));
    }
    vals.sort((a, b) => a - b);
    const sum = vals.reduce((a, b) => a + b, 0);
    stats[`${col.name} — min`] = vals[0]?.toFixed(2);
    stats[`${col.name} — max`] = vals[vals.length - 1]?.toFixed(2);
    stats[`${col.name} — mean`] = (sum / vals.length).toFixed(2);
    stats[`${col.name} — median`] = vals[Math.floor(vals.length / 2)]?.toFixed(2);
  } else if (col.type === DG.COLUMN_TYPE.STRING && col.name.toLowerCase() !== 'smiles' && col.name.toLowerCase() !== 'molecule' && col.name.toLowerCase() !== 'structure') {
    const freq = {};
    for (let r = 0; r < t.rowCount; r++) {
      if (t.filter.get(r)) { const v = col.get(r); freq[v] = (freq[v] || 0) + 1; }
    }
    const top = Object.entries(freq).sort((a, b) => b[1] - a[1]).slice(0, 3).map(([k, v]) => `${k}: ${v}`).join(', ');
    stats[`${col.name} — top values`] = top;
  }
}

return ui.tableFromMap(stats);
